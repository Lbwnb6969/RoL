namespace = rol_relic_fragment

rol_relic_fragment.0001 = {
	type = character_event
	title = rol_relic_fragment.0001.t
	desc = rol_relic_fragment.0001.desc
	theme = learning
	override_background = { reference = rol_bg_fragment_expedition }
	trigger = {
		is_ai = no
		is_alive = yes
		OR = {
			has_character_flag = rol_fragment_route_mars
			has_character_flag = rol_fragment_route_nibelung
			has_character_flag = rol_fragment_route_astrolabe
			has_character_flag = rol_fragment_route_ash
		}
	}
	immediate = {
		if = {
			limit = { has_character_flag = rol_fragment_route_mars }
			random_character_artifact = { limit = { artifact_type = rol_blade_of_mars } save_scope_as = rol_fragment_artifact }
		}
		else_if = {
			limit = { has_character_flag = rol_fragment_route_nibelung }
			random_character_artifact = { limit = { artifact_type = rol_nibelung_star_sphere } save_scope_as = rol_fragment_artifact }
		}
		else_if = {
			limit = { has_character_flag = rol_fragment_route_astrolabe }
			random_character_artifact = { limit = { artifact_type = rol_astrolabe_of_ocean } save_scope_as = rol_fragment_artifact }
		}
		else_if = {
			limit = { has_character_flag = rol_fragment_route_ash }
			random_character_artifact = { limit = { artifact_type = rol_ash_forged_mask } save_scope_as = rol_fragment_artifact }
		}
	}
	option = {
		name = rol_relic_fragment.0001.a
		flavor = rol_relic_fragment.0001.a.flavor
		skill = learning
		random_list = {
			70 = { desc = rol_relic_fragment.0001.success add_learning_lifestyle_xp = 125 hidden_effect = { rol_relic_fragment_success_effect = yes } }
			30 = { desc = rol_relic_fragment.0001.failure hidden_effect = { rol_relic_fragment_failure_effect = yes } }
		}
	}
	option = {
		name = rol_relic_fragment.0001.b
		flavor = rol_relic_fragment.0001.b.flavor
		skill = prowess
		random_list = {
			60 = { desc = rol_relic_fragment.0001.success add_prestige = 50 hidden_effect = { rol_relic_fragment_success_effect = yes } }
			40 = { desc = rol_relic_fragment.0001.failure hidden_effect = { rol_relic_fragment_failure_effect = yes } }
		}
	}
	option = {
		name = rol_relic_fragment.0001.c
		flavor = rol_relic_fragment.0001.c.flavor
		skill = stewardship
		random_list = {
			65 = { desc = rol_relic_fragment.0001.success remove_short_term_gold = 50 hidden_effect = { rol_relic_fragment_success_effect = yes } }
			35 = { desc = rol_relic_fragment.0001.failure hidden_effect = { rol_relic_fragment_failure_effect = yes } }
		}
	}
}
