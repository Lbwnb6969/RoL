	title = rol_quest_munich.1001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:activity
					scope:activity.var:activity_special_type_progression >= 55
				}
				desc = rol_quest_munich.1001.desc.tier_4
			}
			triggered_desc = {
				trigger = {
					exists = scope:activity
					scope:activity.var:activity_special_type_progression >= 35
				}
				desc = rol_quest_munich.1001.desc.partial
			}
			triggered_desc = {
				trigger = {
					OR = {
						NOT = { exists = scope:activity }
						AND = {
							exists = scope:activity
							scope:activity.var:activity_special_type_progression < 55
						}
					}
				}
				desc = rol_quest_munich.1001.desc.below_tier4
			}
			desc = rol_quest_munich.1001.desc
		}
	}
	theme = learning
	override_background = { reference = rol_success_bg_munich }
	override_sound = { reference = "event:/SFX/Events/Themes/sfx_event_theme_type_learning" }

	trigger = {
		rol_munich_quest_active_trigger = yes
		is_ai = no
	}

	immediate = {
		if = {
			limit = { rol_munich_quest_active_trigger = yes }
			if = {
				limit = { has_character_flag = rol_nibelung_sphere_claimed }
				error_log = "[ROL][DUPLICATE] rol_quest_munich.1001 fired after claim; skipping artifact grant"
			}
			else_if = {
				limit = {
					exists = scope:activity
					scope:activity.var:activity_special_type_progression >= 55
				}
				play_sound_effect = "event:/DLC/EP1/SFX/UI/Notifications/court_grandeur_toast_lvl_up"
				rol_create_nibelung_star_sphere_effect = yes

				if = {
					limit = { exists = scope:newly_created_artifact }
					rol_munich_quest_success_finalize_effect = yes
				}
				else = {
					error_log = "[ROL][ERROR] Failed to grant artifact rol_nibelung_star_sphere in rol_quest_munich.1001"
					rol_munich_quest_fail_now_effect = yes
				}

				rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_nibelung_terminal_tier4_reward }
			}
			else_if = {
				limit = { exists = scope:activity scope:activity.var:activity_special_type_progression >= 35 }
				rol_munich_quest_partial_finalize_effect = yes
			}
			else = {
				rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_nibelung_terminal_below_tier4_fail }
				rol_munich_quest_fail_now_effect = yes
			}

		}
	}

	option = {
		name = rol_quest_munich.1001.a

		flavor = rol_quest_munich.1001.a.flavor
		custom_tooltip = rol_quest_munich.1001.a.tt
	}
}
