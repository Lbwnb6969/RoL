rol_quest_munich.1000 = {
	type = character_event
	title = rol_quest_munich.1000.t
	desc = rol_quest_munich.1000.desc
	theme = learning
	override_background = { reference = dungeon }

	trigger = {
		is_ai = no
		has_character_flag = rol_heard_nibelung_legend
		NOT = { has_character_flag = rol_nibelung_sphere_claimed }
	}

	immediate = {
		if = {
			limit = { NOT = { has_variable = rol_munich_stage_count } }
			set_variable = {
				name = rol_munich_stage_count
				value = 0
			}
		}
	}

	option = {
		name = rol_quest_munich.1000.a
		skill = learning
		custom_tooltip = rol_quest_munich.1000.a.base_tt
		if = {
			limit = { learning >= 18 }
			custom_tooltip = rol_quest_munich.1000.a.odds_high_tt
			random_list = {
				70 = {
					add_piety = 100
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_major = yes }
						rol_munich_quest_progress_effect = yes
					}
				}
				30 = {
					add_stress = 40
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_negative = yes }
						rol_munich_quest_progress_effect = yes
					}
				}
			}
		}
		else = {
			custom_tooltip = rol_quest_munich.1000.a.odds_low_tt
			random_list = {
				45 = {
					add_piety = 60
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_medium = yes }
						rol_munich_quest_progress_effect = yes
					}
				}
				55 = {
					add_stress = 50
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_negative_tiny = yes }
						rol_munich_quest_progress_effect = yes
					}
				}
			}
		}
	}

	option = {
		name = rol_quest_munich.1000.b
		skill = martial
		custom_tooltip = rol_quest_munich.1000.b.base_tt
		if = {
			limit = {
				OR = {
					martial >= 16
					prowess >= 14
				}
			}
			custom_tooltip = rol_quest_munich.1000.b.odds_high_tt
			random_list = {
				55 = {
					add_prestige = 90
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_medium = yes }
						rol_munich_quest_progress_effect = yes
					}
				}
				45 = {
					add_trait = wounded_1
					add_prestige = -120
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_negative = yes }
						rol_munich_quest_progress_effect = yes
					}
				}
			}
		}
		else = {
			custom_tooltip = rol_quest_munich.1000.b.odds_low_tt
			random_list = {
				35 = {
					add_prestige = 60
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_tiny = yes }
						rol_munich_quest_progress_effect = yes
					}
				}
				65 = {
					add_trait = wounded_1
					add_prestige = -150
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_negative_major = yes }
						rol_munich_quest_progress_effect = yes
					}
				}
			}
		}
	}

	option = {
		name = rol_quest_munich.1000.c
		custom_tooltip = rol_quest_munich.1000.c.tt
		add_prestige = -50
		hidden_effect = {
			scope:activity ?= { activity_special_type_progression_negative_major = yes }

			rol_apply_nibelung_activity_cooldown_effect = yes
		}
	}
}

				trigger = {

				}
				desc = rol_quest_munich.1001.desc.tier_3
			}
			triggered_desc = {
				trigger = {
					exists = scope:activity
					scope:activity.var:activity_special_type_progression >= 35
				}
				desc = rol_quest_munich.1001.desc.tier_2
			}
			triggered_desc = {
				trigger = { exists = scope:activity }
				desc = rol_quest_munich.1001.desc.tier_1
			}
			desc = rol_quest_munich.1001.desc
		}
	}
	theme = learning
	override_background = { reference = armory }
	override_sound = { reference = "event:/SFX/Events/Themes/sfx_event_theme_type_faith" }

	trigger = {
		is_ai = no
		NOT = { has_character_flag = rol_nibelung_sphere_claimed }
	}

	immediate = {
		play_sound_effect = "event:/DLC/EP1/SFX/UI/Notifications/court_grandeur_toast_lvl_up"

		rol_create_nibelung_star_sphere_effect = yes

		if = {
			limit = { exists = scope:newly_created_artifact }
			scope:newly_created_artifact = {
				set_owner = root
			}
		}
		else = {
			error_log = "[ROL] Failed to grant artifact rol_nibelung_star_sphere in rol_quest_munich.1001"
		}

		rol_munich_quest_success_finalize_effect = yes
	}

	option = {
		name = rol_quest_munich.1001.a
		custom_tooltip = rol_quest_munich.1001.a.tt
	}
}
