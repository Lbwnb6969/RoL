namespace = rol_relic_discovery

# Mythic Relics - 多地点传说发现事件（模板层）

# 入口：隐藏检测事件
rol_relic_discovery.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		is_ai = no
		is_playable_character = yes
		is_alive = yes
		is_landed = yes
	}

	immediate = {
		# 罗马：玛尔斯
		if = {
			limit = {
				has_title = title:c_roma
				NOT = { has_character_flag = rol_heard_mars_legend }
				NOT = { has_character_flag = rol_mars_blade_claimed }
			}
			add_character_flag = rol_heard_mars_legend
			trigger_event = rol_relic_discovery.1000
		}

		# 雅典样例：雅典娜（礼器）
		if = {
			limit = {
				has_title = title:c_attica
				NOT = { has_character_flag = rol_heard_athena_legend }
				NOT = { has_character_flag = rol_athena_relic_claimed }
			}
			add_character_flag = rol_heard_athena_legend
			trigger_event = rol_relic_discovery.1100
		}

		# 吉萨样例：阿努比斯（配饰）
		if = {
			limit = {
				has_title = title:c_giza
				NOT = { has_character_flag = rol_heard_anubis_legend }
				NOT = { has_character_flag = rol_anubis_relic_claimed }
			}
			add_character_flag = rol_heard_anubis_legend
			trigger_event = rol_relic_discovery.1200
		}

		# 慕尼黑：尼伯龙根星界球（礼器）
		if = {
			limit = {
				has_title = title:c_munchen
				NOT = { has_character_flag = rol_heard_nibelung_legend }
				NOT = { has_character_flag = rol_nibelung_sphere_claimed }
			}
			add_character_flag = rol_heard_nibelung_legend
			trigger_event = rol_relic_discovery.1300
		}
	}
}
