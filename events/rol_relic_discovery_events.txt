namespace = rol_relic_discovery

# Mythic Relics - 圣物发现入口与分支事件

# 年度/拿地发现总入口
rol_relic_discovery.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		is_ai = no
		is_playable_character = yes
		is_alive = yes
		is_landed = yes
	}

	immediate = {
		# 罗马：玛尔斯之剑
		if = {
			limit = {
				has_title = title:c_roma
				NOT = { has_character_flag = rol_heard_mars_legend }
				NOT = { has_character_flag = rol_mars_blade_claimed }
			}
			add_character_flag = rol_heard_mars_legend
			trigger_event = rol_relic_discovery.1000
		}

		# 雅典：雅典娜礼器（预留）
		if = {
			limit = {
				has_title = title:c_attica
				NOT = { has_character_flag = rol_heard_athena_legend }
				NOT = { has_character_flag = rol_athena_relic_claimed }
			}
			add_character_flag = rol_heard_athena_legend
			trigger_event = rol_relic_discovery.1100
		}

		# 吉萨：阿努比斯遗物（预留）
		if = {
			limit = {
				has_title = title:c_giza
				NOT = { has_character_flag = rol_heard_anubis_legend }
				NOT = { has_character_flag = rol_anubis_relic_claimed }
			}
			add_character_flag = rol_heard_anubis_legend
			trigger_event = rol_relic_discovery.1200
		}

		# 慕尼黑：尼伯龙根星界球
		if = {
			limit = {
				has_title = title:c_munchen
				NOT = { has_character_flag = rol_heard_nibelung_legend }
				NOT = { has_character_flag = rol_nibelung_sphere_claimed }
			}
			add_character_flag = rol_heard_nibelung_legend
			trigger_event = rol_relic_discovery.1300
		}

		# 纳克索斯 / 泰拉：地下宫殿传说
		if = {
			limit = {
				OR = {
					has_title = title:c_naxos
					has_title = title:b_thera
				}
				NOT = { has_character_flag = rol_heard_naxos_legend }
				NOT = { has_character_flag = rol_naxos_unit_unlocked }
			}
			add_character_flag = rol_heard_naxos_legend
			trigger_event = rol_relic_discovery.1400
		}

		# 塞维利亚：大洋统御星盘
		if = {
			limit = {
				has_title = title:c_sevilla
				NOT = { has_character_flag = rol_heard_astrolabe_legend }
				NOT = { has_character_flag = rol_astrolabe_of_ocean_claimed }
			}
			add_character_flag = rol_heard_astrolabe_legend
			trigger_event = rol_relic_discovery.1500
		}

		# 突尼斯 / 迦太基：浴火金面
		if = {
			limit = {
				OR = {
					has_title = title:c_tunis
					has_title = title:b_qartajana
				}
				NOT = { has_character_flag = rol_heard_carthage_legend }
				NOT = { has_character_flag = rol_ash_forged_mask_claimed }
			}
			add_character_flag = rol_heard_carthage_legend
			trigger_event = rol_relic_discovery.1600
		}
	}
}
