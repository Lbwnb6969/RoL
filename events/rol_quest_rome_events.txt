# Mythic Relics - 罗马地宫多阶段活动事件链

rol_quest_rome.1000 = {
	type = character_event
	title = rol_quest_rome.1000.t
	desc = rol_quest_rome.1000.desc
	theme = martial
	override_background = { reference = corridor_night }

	trigger = {
		is_ai = no
		has_character_flag = rol_heard_mars_legend
		NOT = { has_character_flag = rol_mars_blade_claimed }
	}

	immediate = {
		if = {
			limit = { NOT = { has_variable = rol_rome_stage_count } }
			set_variable = {
				name = rol_rome_stage_count
				value = 0
			}
		}
	}

	option = {
		name = rol_quest_rome.1000.a
		skill = martial
		custom_tooltip = rol_quest_rome.1000.a.base_tt

		if = {
			limit = {
				OR = {
					martial >= 18
					prowess >= 14
				}
			}
			custom_tooltip = rol_quest_rome.1000.a.odds_high_tt
			random_list = {
				60 = {
					add_prestige = 75
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_major = yes }
						rol_rome_quest_progress_effect = yes
					}
				}
				40 = {
					add_trait = wounded_1
					add_prestige = -100
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_negative = yes }
						rol_rome_quest_progress_effect = yes
					}
				}
			}
		}
		else = {
			custom_tooltip = rol_quest_rome.1000.a.odds_low_tt
			random_list = {
				40 = {
					add_prestige = 50
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_medium = yes }
						rol_rome_quest_progress_effect = yes
					}
				}
				60 = {
					add_trait = wounded_1
					add_prestige = -100
					hidden_effect = {
						scope:activity ?= { activity_special_type_progression_negative = yes }
						rol_rome_quest_progress_effect = yes
					}
				}
			}
		}
	}

	option = {
		name = rol_quest_rome.1000.b
		skill = learning
		trigger = {
			learning >= 15
		}
		custom_tooltip = rol_quest_rome.1000.b.tt
		add_piety = 75
		hidden_effect = {
			scope:activity ?= { activity_special_type_progression_major = yes }
			rol_rome_quest_progress_effect = yes
		}
	}

	option = {
		name = rol_quest_rome.1000.c
		custom_tooltip = rol_quest_rome.1000.c.tt
		add_prestige = -50
		hidden_effect = {
			scope:activity ?= { activity_special_type_progression_negative_major = yes }
			rol_rome_quest_abort_effect = yes
			rol_apply_mars_activity_cooldown_effect = yes
		}
	}
}

rol_quest_rome.1001 = {
	type = character_event
	title = rol_quest_rome.1001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:activity
					scope:activity.var:activity_special_type_progression >= 75
				}
				desc = rol_quest_rome.1001.desc.tier_4
			}
			triggered_desc = {
				trigger = {
					exists = scope:activity
					scope:activity.var:activity_special_type_progression >= 50
				}
				desc = rol_quest_rome.1001.desc.tier_3
			}
			triggered_desc = {
				trigger = {
					exists = scope:activity
					scope:activity.var:activity_special_type_progression >= 25
				}
				desc = rol_quest_rome.1001.desc.tier_2
			}
			triggered_desc = {
				trigger = {
					exists = scope:activity
				}
				desc = rol_quest_rome.1001.desc.tier_1
			}
			desc = rol_quest_rome.1001.desc
		}
	}
	theme = martial
	override_background = { reference = armory }
	override_sound = { reference = "event:/SFX/Events/Themes/sfx_event_theme_type_duty" }

	trigger = {
		is_ai = no
		NOT = { has_character_flag = rol_mars_blade_claimed }
	}

	immediate = {
		play_sound_effect = "event:/DLC/EP1/SFX/UI/Notifications/court_grandeur_toast_lvl_up"

		# ????????????????????????
		create_artifact = {
			name = artifact_rol_blade_of_mars
			description = artifact_rol_blade_of_mars_desc
			creator = root
			type = rol_blade_of_mars
			visuals = rol_blade_of_mars_visual
			rarity = illustrious
			template = general_unique_template
			modifier = rol_blade_of_mars_modifier
			decaying = no
			save_scope_as = newly_created_artifact
		}

		if = {
			limit = { exists = scope:newly_created_artifact }
			scope:newly_created_artifact = {
				set_owner = root
				add_artifact_modifier = rol_relic_tier_legendary_modifier
			}
		}
		else = {
			error_log = "[ROL] Failed to grant artifact rol_blade_of_mars in rol_quest_rome.1001"
		}

		# ???????????????????
		rol_rome_clamp_success_progress_effect = yes
		rol_rome_apply_tiered_bonus_reward_effect = yes

		add_character_flag = rol_mars_blade_claimed
		remove_character_flag = rol_heard_mars_legend
		remove_character_modifier = rol_mars_quest_cooldown_modifier
		rol_rome_quest_abort_effect = yes
	}

	option = {
		name = rol_quest_rome.1001.a
		custom_tooltip = rol_quest_rome.1001.a.tt
	}
}
