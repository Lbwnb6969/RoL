rol_quest_rome.1000 = {

				trigger = { rol_relic_context_is_weapon_trigger = yes }

			triggered_desc = {
				trigger = { rol_relic_context_is_accessory_trigger = yes }
				desc = rol_quest_rome.1000.desc.accessory
			}
			desc = rol_quest_rome.1000.desc.generic
		}
	}
	theme = martial
	override_background = { reference = rol_bg_rome_1000_catacombs }

	trigger = {
		is_ai = no
		has_character_flag = rol_heard_mars_legend
		NOT = { has_character_flag = rol_mars_blade_claimed }
	}

	immediate = {
		if = {
			limit = { NOT = { has_variable = rol_rome_stage_count } }
			set_variable = {
				name = rol_rome_stage_count
				value = 0
			}
		}
	}

	option = {
		name = rol_quest_rome.1000.a
		skill = martial
		custom_tooltip = rol_quest_rome.1000.a.base_tt

		if = {
			limit = { rol_expedition_preparation_low_trigger = yes }

			limit = { rol_expedition_preparation_standard_trigger = yes }

			custom_tooltip = rol_expedition_preparation_current_modifier_supreme_tt
		}
		if = {
			limit = {
				OR = {
					martial >= 25
					prowess >= 25
				}
			}
			custom_tooltip = rol_quest_rome.1000.a.odds_elite_tt
						hidden_effect = { remove_character_flag = rol_expedition_random_success }

				chance = {
					value = 85

					max = 95

					scope:activity ?= { activity_special_type_progression_major = yes }
