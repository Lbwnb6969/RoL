rol_rival.1001 = {
	type = character_event
	title = rol_rival.1001.t
	desc = rol_rival.1001.desc
	theme = diplomacy
	override_background = { reference = rol_bg_rival_pact }
	left_portrait = scope:rol_expedition_rival
	trigger = { is_ai = no rol_rival_pact_choice_trigger = yes }
	immediate = {
		# Initialize legacy pending records without replacing a captured new pact.
		if = { limit = { NOT = { has_variable = rol_rival_pact_generation } } rol_rival_capture_pact_effect = yes }
	}
	option = {
		name = rol_rival.1001.a
		flavor = rol_rival.1001.a.flavor
		trigger = { rol_rival_pact_choice_trigger = yes gold >= 50 }
		if = { limit = { rol_rival_pact_choice_trigger = yes gold >= 50 }
			remove_short_term_gold = 50
			add_prestige = 100
			scope:rol_expedition_rival = { add_opinion = { target = root modifier = rol_expedition_cooperation_opinion } }
			rol_rival_close_selected_pact_effect = yes
		}
	}
	option = {
		name = rol_rival.1001.b
		flavor = rol_rival.1001.b.flavor
		custom_tooltip = rol_rival_remember_grudge_tt
		trigger = { rol_rival_pact_choice_trigger = yes }
		if = { limit = { rol_rival_pact_choice_trigger = yes }
			add_prestige = 150
			rol_rival_remember_relationship_effect = { FLAG = rol_rival_grudge_active }
			scope:rol_expedition_rival = { add_opinion = { target = root modifier = rol_expedition_grudge_opinion } }
			rol_rival_close_selected_pact_effect = yes
		}
	}
	option = {
		name = rol_rival.1001.c
		flavor = rol_rival.1001.c.flavor
		trigger = { rol_rival_pact_choice_trigger = yes }
		if = { limit = { rol_rival_pact_choice_trigger = yes }
			add_stress = -10
			scope:rol_expedition_rival = { add_opinion = { target = root modifier = rol_expedition_respect_opinion } }
			rol_rival_close_selected_pact_effect = yes
		}
	}
	option = {
		name = rol_rival.1001.d
		flavor = rol_rival.1001.d.flavor
		trigger = {
			rol_rival_pact_choice_trigger = yes
			gold >= 75
			OR = {
				diplomacy >= 14
				has_trait = rol_relic_seeker
			}
		}
		if = {
			limit = {
				rol_rival_pact_choice_trigger = yes
				gold >= 75
				OR = {
					diplomacy >= 14
					has_trait = rol_relic_seeker
				}
			}
			remove_short_term_gold = 75
			add_prestige = 75
			set_variable = { name = rol_expedition_trusted_companion value = scope:rol_expedition_rival }
			scope:rol_expedition_rival = { add_opinion = { target = root modifier = rol_expedition_fulfilled_pact_opinion } }
			add_opinion = { target = scope:rol_expedition_rival modifier = rol_expedition_fulfilled_pact_opinion }
			if = {
				limit = { NOT = { has_trait = rol_relic_seeker } }
				add_trait = rol_relic_seeker
			}
			add_trait_xp = { trait = rol_relic_seeker track = rol_negotiation value = 10 }
			rol_rival_close_selected_pact_effect = yes
		}
	}
	option = { name = rol_rival_context_lost_option flavor = rol_rival_context_lost_flavor trigger = { NOT = { rol_rival_pact_choice_trigger = yes } } rol_rival_close_selected_pact_effect = yes }
}
