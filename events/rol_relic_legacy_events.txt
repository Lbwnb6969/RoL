namespace = rol_relic_legacy

# First holder event: the relic asks its owner to embody its legend.
rol_relic_legacy.0001 = {
	type = character_event title = rol_relic_legacy.0001.t desc = rol_relic_legacy.0001.desc theme = stewardship
	trigger = { rol_has_true_artifact_relic_trigger = yes NOT = { has_character_flag = rol_relic_legacy_duty_cooldown } }
	immediate = { random_character_artifact = { limit = { OR = { artifact_type = rol_blade_of_mars artifact_type = rol_nibelung_star_sphere artifact_type = rol_astrolabe_of_ocean artifact_type = rol_ash_forged_mask } } save_scope_as = rol_legacy_artifact } add_character_flag = { flag = rol_relic_legacy_duty_cooldown years = 5 } }
	option = { name = rol_relic_legacy.0001.a flavor = rol_relic_legacy.0001.a.flavor add_prestige = 100 add_stress = 15 }
	option = { name = rol_relic_legacy.0001.b flavor = rol_relic_legacy.0001.b.flavor add_piety = 75 add_stress = -15 }
}

rol_relic_legacy.0003 = {
	type = character_event title = rol_relic_legacy.0003.t desc = rol_relic_legacy.0003.desc theme = diplomacy
	override_background = { reference = rol_bg_relic_legacy_transfer }
	trigger = { exists = scope:rol_legacy_artifact NOT = { has_character_flag = rol_relic_legacy_transition_cooldown } }
	immediate = { add_character_flag = { flag = rol_relic_legacy_transition_cooldown years = 5 } }
	option = {
		name = rol_relic_legacy.0003.a
		flavor = rol_relic_legacy.0003.a.flavor
		add_prestige = 75
		add_character_modifier = { modifier = rol_relic_legacy_public_modifier years = 5 }
	}
	option = {
		name = rol_relic_legacy.0003.b
		flavor = rol_relic_legacy.0003.b.flavor
		add_stress = -10
		add_character_modifier = { modifier = rol_relic_legacy_concealed_modifier years = 5 }
	}
	option = {
		name = rol_relic_legacy.0003.c
		flavor = rol_relic_legacy.0003.c.flavor
		trigger = { gold >= 100 }
		remove_short_term_gold = 100
		add_learning_lifestyle_xp = 150
		add_character_modifier = { modifier = rol_relic_legacy_scholarly_modifier years = 5 }
	}
}
