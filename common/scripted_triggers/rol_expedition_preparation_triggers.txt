# Relics of Legend - Unified expedition preparation triggers

rol_expedition_preparation_low_trigger = {
	exists = scope:activity
	scope:activity = {
		has_activity_option = {
			category = rol_option_expedition_preparation
			option = rol_expedition_preparation_low
		}
	}
}
