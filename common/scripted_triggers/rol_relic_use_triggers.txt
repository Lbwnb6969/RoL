rol_relic_use_ready_trigger = {
 custom_description = {
  text = rol_relic_use_ready_tt
  is_alive = yes
  is_adult = yes
  NOT = { has_character_flag = rol_relic_use_pending }
  NOT = { has_character_flag = rol_relic_use_cooldown }
  gold >= 50
 }
}

rol_relic_use_candidate_owned_trigger = {
 exists = scope:rol_relic_use_candidate
 scope:rol_relic_use_candidate = { artifact_owner = root }
}

# Artifact scope.
rol_relic_has_use_mode_trigger = {
 OR = {
  has_artifact_modifier = rol_relic_use_1_field_modifier
  has_artifact_modifier = rol_relic_use_1_oath_modifier
  has_artifact_modifier = rol_relic_use_2_study_modifier
  has_artifact_modifier = rol_relic_use_2_display_modifier
  has_artifact_modifier = rol_relic_use_3_navigation_modifier
  has_artifact_modifier = rol_relic_use_3_teaching_modifier
  has_artifact_modifier = rol_relic_use_4_dread_modifier
  has_artifact_modifier = rol_relic_use_4_restoration_modifier
 }
}

rol_relic_use_pending_trigger = {
 has_character_flag = rol_relic_use_pending
 has_variable = rol_relic_use_generation
 has_variable = rol_relic_use_mode
}

# Each delayed event carries its own generation. An old event cannot cash a new job.
rol_relic_use_generation_matches_trigger = {
 rol_relic_use_pending_trigger = yes
 exists = scope:rol_relic_use_generation
 scope:rol_relic_use_generation = var:rol_relic_use_generation
}
