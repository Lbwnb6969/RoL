rol_expedition_companion_partner_alive_trigger = {
	has_variable = rol_expedition_companion_partner
	exists = var:rol_expedition_companion_partner
	var:rol_expedition_companion_partner = { is_alive = yes }
}

# Vanilla is_available_adult excludes involved_activity; do not use it for attendees.
rol_expedition_companion_fit_trigger = {
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	NOT = { has_trait = incapable }
}

rol_expedition_companion_present_trigger = {
	exists = scope:activity
	rol_expedition_companion_partner_alive_trigger = yes
	scope:activity = {
		var:rol_expedition_companion_open = 1
		any_attending_character = {
			this = root.var:rol_expedition_companion_partner
			rol_expedition_companion_fit_trigger = yes
		}
	}
}

rol_expedition_companion_scholar_trigger = {
	rol_expedition_companion_present_trigger = yes
	scope:activity = { var:rol_expedition_companion_role = 1 }
}

# Person scope: the scholar's earned research, never the host's expertise.
rol_expedition_scholar_experienced_trigger = {
	has_trait = rol_relic_seeker
	has_trait_xp = { trait = rol_relic_seeker track = rol_research value >= 60 }
}

	has_trait = rol_relic_seeker
	has_trait_xp = { trait = rol_relic_seeker track = rol_negotiation value >= 60 }
}

rol_expedition_envoy_accord_trigger = {
	rol_rival_route_active_trigger = yes
	rol_expedition_companion_envoy_trigger = yes
	exists = scope:rol_rival_encounter_envoy
	scope:rol_rival_encounter_envoy = var:rol_expedition_companion_partner
	scope:rol_rival_encounter_envoy = { rol_expedition_envoy_experienced_trigger = yes }
	prestige >= rol_expedition_envoy_accord_cost_value
	NOT = {
		AND = {
			has_character_flag = rol_rival_cooperation_active
			exists = var:rol_rival_relationship_partner
			var:rol_rival_relationship_partner = scope:rol_expedition_rival
		}
	}
}

rol_expedition_first_trial_trigger = {
	exists = scope:activity
	scope:activity = {
		var:rol_expedition_growth_open = 1
		var:rol_expedition_first_trial_used = 0
	}
	OR = {
		AND = { has_character_flag = rol_relic_context_mars OR = { NOT = { has_variable = rol_rome_stage_count } var:rol_rome_stage_count = 0 } }
		AND = { has_character_flag = rol_relic_context_nibelung OR = { NOT = { has_variable = rol_munich_stage_count } var:rol_munich_stage_count = 0 } }
		AND = { has_character_flag = rol_relic_context_naxos OR = { NOT = { has_variable = rol_naxos_stage_count } var:rol_naxos_stage_count = 0 } }
		AND = { has_character_flag = rol_relic_context_astrolabe OR = { NOT = { has_variable = rol_sevilla_stage_count } var:rol_sevilla_stage_count = 0 } }
		AND = { has_character_flag = rol_relic_context_carthage OR = { NOT = { has_variable = rol_carthage_stage_count } var:rol_carthage_stage_count = 0 } }
	}
}

rol_expedition_expert_route_trigger = {
	rol_expedition_first_trial_trigger = yes
	has_trait = rol_relic_seeker
	OR = {
		has_trait_xp = { trait = rol_relic_seeker track = rol_research value >= 60 }
		has_trait_xp = { trait = rol_relic_seeker track = rol_exploration value >= 60 }
		has_trait_xp = { trait = rol_relic_seeker track = rol_negotiation value >= 60 }
	}
}

# Evaluated on a potential rival, with the original host saved by the selector.
rol_expedition_rival_candidate_trigger = {
	this != scope:rol_expedition_host
	rol_expedition_companion_fit_trigger = yes
	trigger_if = {
		limit = { exists = scope:rol_expedition_host.var:rol_expedition_companion_partner }
		this != scope:rol_expedition_host.var:rol_expedition_companion_partner
	}
}

 rol_expedition_companion_pending_trigger = yes
 exists = scope:rol_expedition_companion_generation
 scope:rol_expedition_companion_generation = var:rol_expedition_companion_generation
}

# A master may replace a personally earned guard's cash fee with public credit.
# The original person and settlement generation must still match this window.
rol_expedition_companion_master_credit_trigger = {
	rol_expedition_companion_settlement_choice_trigger = yes
	rol_expedition_companion_partner_alive_trigger = yes
	exists = scope:rol_expedition_companion
	scope:rol_expedition_companion = var:rol_expedition_companion_partner
	var:rol_expedition_companion_role = 2
	has_variable = rol_expedition_companion_earned_xp
	var:rol_expedition_companion_earned_xp > 0
	has_trait = rol_relic_seeker
	has_trait_xp = { trait = rol_relic_seeker track = rol_negotiation value >= 100 }
	prestige >= 100
}
