# Mythic Relics - 纳克索斯活动推进逻辑
# 变量：rol_naxos_stage_count（角色作用域）

rol_naxos_quest_progress_effect = {
	if = {
		limit = { NOT = { has_variable = rol_naxos_stage_count } }
		set_variable = {
			name = rol_naxos_stage_count
			value = 0
		}
	}

	change_variable = {
		name = rol_naxos_stage_count
		add = 1
	}

	rol_relic_log_stage_effect = yes
	rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_naxos_stage_progress }

	# 宾客互动事件（每次活动至多触发一次）
	if = {
		limit = {
			NOT = { has_character_flag = rol_naxos_guest_event_fired }
			exists = scope:activity
			scope:activity = {
				any_attending_character = {
					this != root
				}
			}
			var:rol_naxos_stage_count >= 1
			var:rol_naxos_stage_count < 3
		}
		add_character_flag = rol_naxos_guest_event_fired
		trigger_event = { id = rol_quest_naxos.1050 days = 4 }
	}

	if = {
		limit = { var:rol_naxos_stage_count < 3 }
		random_list = {
			50 = { trigger_event = { id = rol_quest_naxos.1002 days = 12 } }
			50 = { trigger_event = { id = rol_quest_naxos.1003 days = 16 } }
		}
	}
	else = {
		trigger_event = { id = rol_quest_naxos.1001 days = 8 }
	}
}

rol_naxos_clamp_success_progress_effect = {
	scope:activity ?= {
		if = {
			limit = { var:activity_special_type_progression < 0 }
			set_variable = {
				name = activity_special_type_progression
				value = 0
			}
		}
		else_if = {
			limit = { var:activity_special_type_progression > 100 }
			set_variable = {
				name = activity_special_type_progression
				value = 100
			}
		}
	}
}

rol_naxos_apply_tiered_bonus_reward_effect = {
	if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 75
		}
		add_prestige = 220
		add_gold = 80
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 50
		}
		add_prestige = 150
		add_gold = 50
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 25
		}
		add_prestige = 90
	}
	else = {
		add_prestige = 40
	}
}

rol_naxos_store_success_result_effect = {
	if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 75
		}
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 4
		}
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 50
		}
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 3
		}
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 25
		}
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 2
		}
	}
	else = {
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 1
		}
	}

	add_character_flag = rol_recent_naxos_quest_success
	remove_character_flag = rol_recent_naxos_quest_failure
}

rol_naxos_store_failure_result_effect = {
	set_variable = {
		name = rol_recent_relic_result_tier
		value = 0
	}
	add_character_flag = rol_recent_naxos_quest_failure
	remove_character_flag = rol_recent_naxos_quest_success
}

rol_naxos_quest_success_finalize_effect = {
	rol_naxos_clamp_success_progress_effect = yes
	rol_naxos_store_success_result_effect = yes
	rol_naxos_apply_tiered_bonus_reward_effect = yes
	rol_apply_expedition_preparation_success_bonus_effect = yes

	add_character_flag = rol_naxos_unit_unlocked
	rol_check_legend_circle_revelation_effect = yes
	remove_character_flag = rol_heard_naxos_legend
	remove_character_modifier = rol_naxos_quest_cooldown_modifier

	rol_relic_log_success_effect = yes
	send_interface_toast = {
		type = event_toast_effect_good
		title = rol_naxos_activity_success_toast_title
		desc = rol_naxos_activity_success_toast_desc
		left_icon = root
	}

	rol_clear_context_naxos_relic_effect = yes
	rol_naxos_quest_abort_effect = yes
	rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_naxos_success_finalize }
}

rol_apply_naxos_activity_cooldown_effect = {
	rol_relic_apply_cooldown_effect = {
		COOLDOWN_MODIFIER = rol_naxos_quest_cooldown_modifier
		COOLDOWN_YEARS = 2
	}

	rol_naxos_store_failure_result_effect = yes
	rol_relic_log_failure_effect = yes
	send_interface_toast = {
		type = event_toast_effect_bad
		title = rol_naxos_activity_failure_toast_title
		desc = rol_naxos_activity_failure_toast_desc
		left_icon = root
	}

	rol_clear_context_naxos_relic_effect = yes
	rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_naxos_failure_finalize }
}

rol_naxos_quest_fail_now_effect = {
	rol_apply_naxos_activity_cooldown_effect = yes
	rol_naxos_quest_abort_effect = yes
	rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_naxos_fail_now }
	scope:activity ?= {
		complete_activity = yes
	}
}

rol_naxos_quest_abort_effect = {
	if = {
		limit = { has_variable = rol_naxos_stage_count }
		remove_variable = rol_naxos_stage_count
	}
}
