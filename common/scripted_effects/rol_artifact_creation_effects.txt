# Mythic Relics - 测试神器创建效果
# 调用方式（角色作用域）：rol_create_sword_of_mars_effect = yes
rol_create_sword_of_mars_effect = {
	create_artifact = {
		name = rol_sword_of_mars_name
		description = rol_sword_of_mars_description
		creator = root
		type = rol_sword_of_mars
		visuals = sword
		rarity = illustrious
		template = general_unique_template
		modifier = rol_sword_of_mars_modifier
		decaying = no
		save_scope_as = newly_created_artifact
	}

	if = {
		limit = { exists = scope:newly_created_artifact }
		scope:newly_created_artifact = { set_owner = root }
	}
}
