# Called only by the actual host's activity on_start, after route context setup.
rol_expedition_growth_begin_effect = {
	if = {
		limit = { exists = scope:activity }
		remove_character_flag = rol_expedition_remedy_pending
		scope:activity = {
			set_variable = { name = rol_expedition_growth_open value = 1 }
			set_variable = { name = rol_expedition_journal_bonus value = 0 }
			set_variable = { name = rol_expedition_remedy_used value = 0 }
		}
		if = {
			limit = { has_character_modifier = rol_expedition_$ROUTE$_notes_modifier }
			remove_character_modifier = rol_expedition_$ROUTE$_notes_modifier
			scope:activity = { set_variable = { name = rol_expedition_journal_bonus value = 5 } }
			send_interface_toast = {
				type = event_toast_effect_good
				title = rol_expedition_journal_used_title
				desc = rol_expedition_journal_used_desc
				left_icon = this
			}
		}
	}
}

# RESULT_XP: 30 full / 20 partial / 10 failure with completed stages.
# The activity owns the replay guard; direct finalize commands cannot farm XP.
rol_expedition_growth_finish_effect = {
	if = {
		limit = {
			exists = scope:activity
			scope:activity = { var:rol_expedition_growth_open = 1 }
		}
		scope:activity = { set_variable = { name = rol_expedition_growth_open value = 0 } }
		if = {
			limit = { always = $LEAVE_NOTES$ }
			remove_character_modifier = rol_expedition_$ROUTE$_notes_modifier
			add_character_modifier = { modifier = rol_expedition_$ROUTE$_notes_modifier years = 10 }
		}
		if = {
			limit = { NOT = { has_variable = rol_expedition_growth_pending_xp } }
			set_variable = { name = rol_expedition_growth_pending_xp value = 0 }
		}
		change_variable = { name = rol_expedition_growth_pending_xp add = $RESULT_XP$ }
		if = {
			limit = { NOT = { has_character_flag = rol_expedition_growth_pending } }
			add_character_flag = rol_expedition_growth_pending
			trigger_event = { id = rol_expedition_growth.0001 days = 2 }
		}
	}
}

# An abandoned expedition without completed stages gives no expertise.
rol_expedition_growth_abandon_effect = {
	scope:activity ?= { set_variable = { name = rol_expedition_growth_open value = 0 } }
}

# The same guard covers all reflection choices, including manual event replay.
rol_expedition_growth_choose_focus_effect = {
	if = {
		limit = { rol_expedition_growth_pending_trigger = yes }
		if = {
			limit = { NOT = { has_trait = rol_relic_seeker } }
			add_trait = rol_relic_seeker
		}
		add_trait_xp = { trait = rol_relic_seeker track = $TRACK$ value = var:rol_expedition_growth_pending_xp }
		remove_character_modifier = rol_expedition_research_focus_modifier
		remove_character_modifier = rol_expedition_exploration_focus_modifier
		remove_character_modifier = rol_expedition_negotiation_focus_modifier
		add_character_modifier = { modifier = $FOCUS_MODIFIER$ years = 3 }
		if = { limit = { has_trait = $MATCHING_TRAIT$ } add_stress = -5 }
		remove_character_flag = rol_expedition_growth_pending
		remove_variable = rol_expedition_growth_pending_xp
	}
}

# Explicit test helper: queues 20 reflection XP once, never creates an artifact.
rol_expedition_growth_debug_queue_effect = {
	if = {
		limit = { NOT = { rol_expedition_growth_pending_trigger = yes } }
		set_variable = { name = rol_expedition_growth_pending_xp value = 20 }
		add_character_flag = rol_expedition_growth_pending
		trigger_event = rol_expedition_growth.0001
	}
}

# Record only the loss caused by this roll; later interludes keep their own progress.
rol_expedition_record_trial_failure_effect = {
	scope:activity = { set_variable = { name = rol_expedition_trial_failure_loss value = $LOSS$ } }
}

rol_expedition_capture_remedy_window_effect = {
	if = {
		limit = { NOT = { has_variable = rol_expedition_remedy_generation } }
		set_variable = { name = rol_expedition_remedy_generation value = 1 }
	}
	save_scope_value_as = { name = rol_expedition_remedy_generation value = var:rol_expedition_remedy_generation }
}

rol_expedition_close_remedy_window_effect = {
	if = {
		limit = {
			exists = scope:rol_expedition_remedy_generation
			scope:rol_expedition_remedy_generation = var:rol_expedition_remedy_generation
		}
		remove_character_flag = rol_expedition_remedy_pending
	}
}

# Recover the current host's pending crisis after a legacy/unavailable window.
rol_expedition_review_remedy_effect = {
	if = {
		limit = { has_character_flag = rol_expedition_remedy_pending exists = involved_activity }
		save_scope_as = host
		involved_activity = { save_scope_as = activity }
		if = {
			limit = {
				rol_relic_any_active_quest_trigger = yes
				scope:activity = { var:rol_expedition_growth_open = 1 var:rol_expedition_remedy_used = 1 }
			}
			rol_expedition_capture_remedy_window_effect = yes
			trigger_event = rol_expedition_growth.0010
		}
	}
}

# Intercepts a stage roll failure once per expedition. Only experts, consumed
# regional notes or an attending trusted partner can preserve a full-relic route.
rol_expedition_try_remedy_or_fail_effect = {
	if = {
		limit = {
			rol_$CHAIN$_quest_active_trigger = yes
			rol_expedition_can_remedy_trigger = yes
		}
		scope:activity = { set_variable = { name = rol_expedition_remedy_used value = 1 } }
		if = {
			limit = { NOT = { has_variable = rol_expedition_remedy_generation } }
			set_variable = { name = rol_expedition_remedy_generation value = 0 }
		}
		change_variable = { name = rol_expedition_remedy_generation add = 1 }
		rol_expedition_capture_remedy_window_effect = yes
		add_character_flag = rol_expedition_remedy_pending
		trigger_event = { id = rol_expedition_growth.0010 days = 1 }
	}
	else_if = {
		limit = { rol_$CHAIN$_quest_active_trigger = yes }
		rol_$CHAIN$_quest_fail_now_effect = yes
	}
}

rol_expedition_remedy_press_on_effect = {
	if = {
		limit = {
			rol_expedition_remedy_window_valid_trigger = yes
			rol_expedition_remedy_press_on_qualified_trigger = yes
			scope:activity = { has_variable = rol_expedition_trial_failure_loss }
			gold >= 50
		}
		remove_character_flag = rol_expedition_remedy_pending
		remove_short_term_gold = 50
		add_stress = 10
		scope:activity = {
			change_variable = { name = activity_special_type_progression add = var:rol_expedition_trial_failure_loss }
			activity_special_type_progression_major = yes
		}
		if = { limit = { rol_rome_quest_active_trigger = yes } rol_rome_quest_progress_effect = yes }
		else_if = { limit = { rol_munich_quest_active_trigger = yes } rol_munich_quest_progress_effect = yes }
		else_if = { limit = { rol_naxos_quest_active_trigger = yes } rol_naxos_quest_progress_effect = yes }
		else_if = { limit = { rol_sevilla_quest_active_trigger = yes } rol_sevilla_quest_progress_effect = yes }
		else_if = { limit = { rol_carthage_quest_active_trigger = yes } rol_carthage_quest_progress_effect = yes }
	}
}

rol_expedition_remedy_salvage_partial_effect = {
	if = {
		limit = { rol_expedition_remedy_window_valid_trigger = yes }
		remove_character_flag = rol_expedition_remedy_pending
		scope:activity = {
			if = {
				limit = { var:activity_special_type_progression < 35 }
				set_variable = { name = activity_special_type_progression value = 35 }
			}
		}
		if = { limit = { rol_rome_quest_active_trigger = yes } rol_rome_quest_partial_finalize_effect = yes }
		else_if = { limit = { rol_munich_quest_active_trigger = yes } rol_munich_quest_partial_finalize_effect = yes }
		else_if = { limit = { rol_naxos_quest_active_trigger = yes } rol_naxos_quest_partial_finalize_effect = yes }
		else_if = { limit = { rol_sevilla_quest_active_trigger = yes } rol_sevilla_quest_partial_finalize_effect = yes }
		else_if = { limit = { rol_carthage_quest_active_trigger = yes } rol_carthage_quest_partial_finalize_effect = yes }
	}
}

rol_expedition_remedy_retreat_effect = {
	if = {
		limit = { rol_expedition_remedy_window_valid_trigger = yes }
		remove_character_flag = rol_expedition_remedy_pending
		add_stress = -15
		if = {
			limit = { has_trait = rol_relic_seeker }
			add_trait_xp = { trait = rol_relic_seeker track = rol_exploration value = 5 }
		}
		if = { limit = { rol_rome_quest_active_trigger = yes } rol_rome_quest_fail_now_effect = yes }
		else_if = { limit = { rol_munich_quest_active_trigger = yes } rol_munich_quest_fail_now_effect = yes }
		else_if = { limit = { rol_naxos_quest_active_trigger = yes } rol_naxos_quest_fail_now_effect = yes }
		else_if = { limit = { rol_sevilla_quest_active_trigger = yes } rol_sevilla_quest_fail_now_effect = yes }
		else_if = { limit = { rol_carthage_quest_active_trigger = yes } rol_carthage_quest_fail_now_effect = yes }
	}
}
