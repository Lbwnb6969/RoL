# Relics of Legends - artifact provenance & multi-relic resonance bookkeeping

# Explicit test entry: retain the item scope across dispatch.
rol_relic_legacy_debug_transition_effect = {
	if = {
		limit = { rol_has_true_artifact_relic_trigger = yes NOT = { has_character_flag = rol_relic_legacy_transition_cooldown } }
		random_character_artifact = {
			limit = { OR = { artifact_type = rol_blade_of_mars artifact_type = rol_nibelung_star_sphere artifact_type = rol_astrolabe_of_ocean artifact_type = rol_ash_forged_mask } }
			save_scope_as = rol_legacy_artifact
		}
		trigger_event = rol_relic_legacy.0003
	}
}

rol_relic_legacy_register_new_artifact_effect = {
	if = {
		limit = { exists = scope:newly_created_artifact }
		scope:newly_created_artifact = {
			set_variable = {
				name = rol_relic_legacy_transfer_count
				value = 0
			}
		}
	}
}

rol_relic_legacy_record_transfer_effect = {
	if = {
		limit = { exists = scope:rol_legacy_artifact }
		scope:rol_legacy_artifact = {
			if = {
				limit = { has_variable = rol_relic_legacy_transfer_count }
				change_variable = { name = rol_relic_legacy_transfer_count add = 1 }
			}
			else = {
				set_variable = { name = rol_relic_legacy_transfer_count value = 1 }
			}
		}
	}
}

# Each annual window owns a captured generation. A five-year-old window cannot
# spend a later event's qualification, even if the same relic is still owned.
rol_legacy_recurring_begin_effect = {
	if = {
		limit = { NOT = { has_variable = rol_legacy_$NAME$_generation } }
		set_variable = { name = rol_legacy_$NAME$_generation value = 0 }
	}
	change_variable = { name = rol_legacy_$NAME$_generation add = 1 }
	save_scope_value_as = { name = rol_legacy_$NAME$_generation value = var:rol_legacy_$NAME$_generation }
	add_character_flag = rol_legacy_$NAME$_choice_open
}

rol_legacy_recurring_close_effect = {
	if = {
		limit = {
			has_character_flag = rol_legacy_$NAME$_choice_open
			exists = scope:rol_legacy_$NAME$_generation
			scope:rol_legacy_$NAME$_generation = var:rol_legacy_$NAME$_generation
		}
		remove_character_flag = rol_legacy_$NAME$_choice_open
	}
}

rol_clear_relic_resonance_modifiers_effect = {
	remove_character_modifier = rol_resonance_imperial_might_modifier
	remove_character_modifier = rol_resonance_cosmic_chart_modifier
	remove_character_modifier = rol_resonance_sovereign_awe_modifier
}

rol_relic_resonance_begin_effect = {
	if = {
		limit = {
			rol_relic_count_at_least_two_trigger = yes
			NOT = { has_character_flag = rol_relic_resonance_cooldown }
		}
		add_character_flag = { flag = rol_relic_resonance_cooldown years = 5 }
		rol_legacy_recurring_begin_effect = { NAME = resonance }
		trigger_event = rol_relic_legacy.0010
	}
}

rol_relic_resonance_apply_effect = {
	if = {
		limit = { rol_legacy_resonance_choice_trigger = yes }
		rol_clear_relic_resonance_modifiers_effect = yes
		add_character_modifier = { modifier = $MODIFIER$ years = 10 }
		if = {
			limit = { NOT = { has_trait = rol_relic_seeker } }
			add_trait = rol_relic_seeker
		}
		add_trait_xp = { trait = rol_relic_seeker track = $TRACK$ value = 15 }
		rol_legacy_recurring_close_effect = { NAME = resonance }
	}
}
