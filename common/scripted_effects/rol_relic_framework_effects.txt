# Mythic Relics - 多地点神兵活动通用模板

# 设定上下文（角色作用域）
rol_relic_set_context_effect = {
	add_character_flag = rol_relic_context_active
	add_character_flag = $RELIC_CONTEXT_FLAG$
	add_character_flag = $RELIC_ARCHETYPE_FLAG$
}

# 清理上下文（角色作用域）
rol_relic_clear_context_effect = {
	remove_character_flag = rol_relic_context_active
	remove_character_flag = $RELIC_CONTEXT_FLAG$
	remove_character_flag = $RELIC_ARCHETYPE_FLAG$
}

# 通用冷却（角色作用域）
rol_relic_apply_cooldown_effect = {
	add_character_modifier = {
		modifier = $COOLDOWN_MODIFIER$
		years = $COOLDOWN_YEARS$
	}
}

# DEBUG开关（角色作用域）
rol_relic_debug_enable_effect = {
	add_character_flag = rol_relic_debug_mode
	send_interface_toast = {
		type = event_toast_effect_good
		title = rol_relic_debug_enabled_toast
		left_icon = root
	}
}

rol_relic_debug_disable_effect = {
	remove_character_flag = rol_relic_debug_mode
	send_interface_toast = {
		type = event_toast_effect_good
		title = rol_relic_debug_disabled_toast
		left_icon = root
	}
}

# DEBUG追踪（角色作用域）
# 调用示例：
# rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_mars_stage_progress }
rol_relic_debug_trace_effect = {
	if = {
		limit = { has_character_flag = rol_relic_debug_mode }
		send_interface_toast = {
			type = event_toast_effect_good
			title = $DEBUG_KEY$
			left_icon = root
		}
		error_log = "[ROL][DEBUG] $DEBUG_KEY$"
	}
}

# 活动日志：阶段推进
rol_relic_log_stage_effect = {
	scope:activity ?= {
		add_activity_log_entry = {
			key = rol_relic_activity_stage_log
			character = root
			score = 20
			show_in_conclusion = yes
		}
	}
}

# 活动日志：成功
rol_relic_log_success_effect = {
	scope:activity ?= {
		if = {
			limit = { exists = scope:newly_created_artifact }
			add_activity_log_entry = {
				key = rol_relic_activity_success_log
				character = root
				artifact = scope:newly_created_artifact
				score = 90
				tags = { complete good }
				show_in_conclusion = yes
			}
		}
		else = {
			add_activity_log_entry = {
				key = rol_relic_activity_success_log
				character = root
				score = 90
				tags = { complete good }
				show_in_conclusion = yes
			}
		}
	}
}

# 活动日志：失败
rol_relic_log_failure_effect = {
	scope:activity ?= {
		add_activity_log_entry = {
			key = rol_relic_activity_failure_log
			character = root
			score = 70
			tags = { complete bad }
			show_in_conclusion = yes
		}
	}
}

# 清理临时结算状态（角色作用域）
rol_relic_clear_recent_result_effect = {
	remove_character_flag = rol_recent_mars_quest_success
	remove_character_flag = rol_recent_mars_quest_failure
	remove_character_flag = rol_recent_nibelung_quest_success
	remove_character_flag = rol_recent_nibelung_quest_failure
	remove_character_flag = rol_recent_naxos_quest_success
	remove_character_flag = rol_recent_naxos_quest_failure
	remove_character_flag = rol_recent_sevilla_quest_success
	remove_character_flag = rol_recent_sevilla_quest_failure
	remove_character_flag = rol_recent_carthage_quest_success
	remove_character_flag = rol_recent_carthage_quest_failure
	remove_character_flag = rol_expedition_random_success
	if = {
		limit = { has_variable = rol_recent_relic_result_tier }
		remove_variable = rol_recent_relic_result_tier
	}
}
