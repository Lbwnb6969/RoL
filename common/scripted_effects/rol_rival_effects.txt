rol_try_rival_interlude_effect = {
	if = {
		limit = { has_character_flag = rol_rival_cooperation_active }
		random_list = {
			75 = { rol_fire_rival_or_fallback_effect = { FALLBACK_EVENT = $FALLBACK_EVENT$ } }
			25 = { trigger_event = { id = $FALLBACK_EVENT$ days = 4 } rol_clear_rival_route_effect = yes }
		}
	}
	else_if = {
		limit = { has_character_flag = rol_rival_grudge_active }
		random_list = {
			70 = { rol_fire_rival_or_fallback_effect = { FALLBACK_EVENT = $FALLBACK_EVENT$ } }
			30 = { trigger_event = { id = $FALLBACK_EVENT$ days = 4 } rol_clear_rival_route_effect = yes }
		}
	}
	else = {
		random_list = {
			40 = { rol_fire_rival_or_fallback_effect = { FALLBACK_EVENT = $FALLBACK_EVENT$ } }
			60 = { trigger_event = { id = $FALLBACK_EVENT$ days = 4 } rol_clear_rival_route_effect = yes }
		}
	}
}

rol_select_expedition_rival_effect = {
	save_scope_as = rol_expedition_host
	scope:activity = {
		random_attending_character = {
			limit = { this != scope:rol_expedition_host is_alive = yes is_adult = yes }
			save_scope_as = rol_expedition_rival
		}
	}
	if = {
		limit = { NOT = { exists = scope:rol_expedition_rival } }
		random_courtier_or_guest = {
			limit = { this != root is_alive = yes is_adult = yes }
			save_scope_as = rol_expedition_rival
		}
	}
}

rol_fire_rival_or_fallback_effect = {
	rol_select_expedition_rival_effect = yes
	if = {
		limit = { exists = scope:rol_expedition_rival }
		trigger_event = { id = rol_rival.1000 days = 4 }
	}
	else = {
		trigger_event = { id = $FALLBACK_EVENT$ days = 4 }
		rol_clear_rival_route_effect = yes
	}
}

# A convincing counterfeit ends this expedition, but does not set the unique-relic claim flag.
rol_rival_counterfeit_failure_effect = {
	if = { limit = { has_character_flag = rol_rival_route_mars } rol_rome_quest_fail_now_effect = yes }
	else_if = { limit = { has_character_flag = rol_rival_route_nibelung } rol_munich_quest_fail_now_effect = yes }
	else_if = { limit = { has_character_flag = rol_rival_route_astrolabe } rol_sevilla_quest_fail_now_effect = yes }
	else_if = { limit = { has_character_flag = rol_rival_route_carthage } rol_carthage_quest_fail_now_effect = yes }
	else = { rol_naxos_quest_fail_now_effect = yes }
}

rol_clear_rival_route_effect = {

	remove_character_flag = rol_rival_route_nibelung
	remove_character_flag = rol_rival_route_astrolabe
	remove_character_flag = rol_rival_route_carthage
	remove_character_flag = rol_rival_route_naxos
}
