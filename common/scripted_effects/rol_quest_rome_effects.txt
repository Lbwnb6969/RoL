# Mythic Relics - 罗马地宫任务链推进逻辑
# 变量：rol_rome_stage_count（角色作用域）

rol_rome_quest_progress_effect = {
	if = {
		limit = { NOT = { has_variable = rol_rome_stage_count } }
		set_variable = {
			name = rol_rome_stage_count
			value = 0
		}
	}

	change_variable = {
		name = rol_rome_stage_count
		add = 1
	}

	# 记录阶段推进（活动结算页可见）
	rol_relic_log_stage_effect = yes
	rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_mars_stage_progress }

	# 宾客互动事件（每次活动至多触发一次）
	if = {
		limit = {
			NOT = { has_character_flag = rol_mars_guest_event_fired }
			exists = scope:activity
			scope:activity = {
				any_attending_character = {
					this != root
				}
			}
			var:rol_rome_stage_count >= 1
			var:rol_rome_stage_count < 4
		}
		add_character_flag = rol_mars_guest_event_fired
		add_character_flag = rol_rival_route_mars
		rol_try_rival_interlude_effect = { FALLBACK_EVENT = rol_quest_rome.1050 }
	}

	# 按固定顺序推进，避免重复或倒序触发叙事阶段
	if = {
		limit = { var:rol_rome_stage_count = 1 }
		trigger_event = { id = rol_quest_rome.1002 days = 12 }
	}
	else_if = {
		limit = { var:rol_rome_stage_count = 2 }
		trigger_event = { id = rol_quest_rome.1003 days = 16 }
	}
	else_if = {
		limit = { var:rol_rome_stage_count = 3 }
		trigger_event = { id = rol_quest_rome.1004 days = 20 }
	}
	else = {
		trigger_event = { id = rol_quest_rome.1001 days = 10 }
	}
}

# 将活动成功条限制在 0~100，避免超出 UI 预期
rol_rome_clamp_success_progress_effect = {
	scope:activity ?= {
		if = {
			limit = { var:activity_special_type_progression < 0 }
			set_variable = {
				name = activity_special_type_progression
				value = 0
			}
		}
		else_if = {
			limit = { var:activity_special_type_progression > 100 }
			set_variable = {
				name = activity_special_type_progression
				value = 100
			}
		}
	}
}

# 终局额外奖励（不影响神器发放）
rol_rome_apply_tiered_bonus_reward_effect = {
	if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 70
		}
		add_prestige = 250
		add_piety = 150
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 50
		}
		add_prestige = 150
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 30
		}
		add_prestige = 75
	}
	else = {
		add_prestige = 25
	}
}

# 存储最近一次成功结果等级（用于活动结算事件）
rol_rome_store_success_result_effect = {
	if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 70
		}
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 4
		}
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 50
		}
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 3
		}
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 30
		}
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 2
		}
	}
	else = {
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 1
		}
	}

	add_character_flag = rol_recent_mars_quest_success
	remove_character_flag = rol_recent_mars_quest_failure
}

# 存储最近一次失败结果（用于活动结算事件）
rol_rome_store_failure_result_effect = {
	set_variable = {
		name = rol_recent_relic_result_tier
		value = 0
	}
	add_character_flag = rol_recent_mars_quest_failure
	remove_character_flag = rol_recent_mars_quest_success
}

# 罗马远征成功后的统一收尾
rol_rome_quest_success_finalize_effect = {
	rol_expedition_growth_finish_effect = { ROUTE = mars RESULT_XP = 30 LEAVE_NOTES = no }
	rol_rome_clamp_success_progress_effect = yes
	rol_rome_store_success_result_effect = yes
	rol_rome_apply_tiered_bonus_reward_effect = yes
	rol_apply_expedition_preparation_success_bonus_effect = yes
	rol_store_recent_expedition_preparation_effect = yes

	add_character_flag = rol_mars_blade_claimed
	rol_check_legend_circle_revelation_effect = yes
	rol_check_legend_circle_fivefold_revelation_effect = yes
	remove_character_flag = rol_heard_mars_legend
	remove_character_modifier = rol_mars_quest_cooldown_modifier

	rol_relic_log_success_effect = yes
	send_interface_toast = {
		type = event_toast_effect_good
		title = rol_mars_activity_success_toast_title
		desc = rol_mars_activity_success_toast_desc
		left_icon = root
	}

	trigger_event = { id = rol_quest_rome.1090 days = 1 }

	rol_clear_context_mars_weapon_effect = yes
	rol_rome_quest_abort_effect = yes
	rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_mars_success_finalize }
}

rol_rome_quest_partial_finalize_effect = {
	rol_expedition_growth_finish_effect = { ROUTE = mars RESULT_XP = 20 LEAVE_NOTES = yes }
	rol_rome_clamp_success_progress_effect = yes
	rol_rome_store_success_result_effect = yes
	rol_rome_apply_tiered_bonus_reward_effect = yes
	rol_apply_expedition_preparation_success_bonus_effect = yes
	rol_store_recent_expedition_preparation_effect = yes
	rol_relic_partial_success_finalize_effect = {
		PARTIAL_MODIFIER = rol_partial_relic_evidence_modifier
		COOLDOWN_MODIFIER = rol_mars_quest_cooldown_modifier
		SUMMARY_EVENT = rol_quest_rome.1090
		DEBUG_KEY = rol_relic_debug_mars_partial_finalize
	}
	rol_clear_context_mars_weapon_effect = yes
	rol_rome_quest_abort_effect = yes
}

rol_apply_mars_activity_cooldown_effect = {
	if = {
		limit = { has_variable = rol_rome_stage_count var:rol_rome_stage_count >= 1 }
		rol_expedition_growth_finish_effect = { ROUTE = mars RESULT_XP = 10 LEAVE_NOTES = no }
	}
	else = { rol_expedition_growth_abandon_effect = yes }
	rol_relic_apply_cooldown_effect = {
		COOLDOWN_MODIFIER = rol_mars_quest_cooldown_modifier
		COOLDOWN_YEARS = 2
	}

	rol_rome_store_failure_result_effect = yes
	rol_relic_log_failure_effect = yes
	send_interface_toast = {
		type = event_toast_effect_bad
		title = rol_mars_activity_failure_toast_title
		desc = rol_mars_activity_failure_toast_desc
		left_icon = root
	}

	trigger_event = { id = rol_quest_rome.1090 days = 1 }

	rol_clear_context_mars_weapon_effect = yes
	rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_mars_failure_finalize }
}

# 任一失败立即终止活动（统一收口）
rol_rome_quest_fail_now_effect = {
	rol_apply_mars_activity_cooldown_effect = yes
	rol_rome_quest_abort_effect = yes
	rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_mars_fail_now }
}

rol_rome_quest_abort_effect = {
	if = {
		limit = { has_variable = rol_rome_stage_count }
		remove_variable = rol_rome_stage_count
	}
}
