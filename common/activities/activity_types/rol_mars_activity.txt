# Mythic Relics - 战神玛尔斯神兵活动
activity_rol_seek_mars_blade = {
	is_shown = {
		is_playable_character = yes
		is_ai = no
		has_character_flag = rol_heard_mars_legend
		NOT = { has_character_flag = rol_mars_blade_claimed }
	}

	can_start_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
		gold >= 180
		NOT = { is_activity_type_on_cooldown = activity_rol_seek_mars_blade }
		custom_tooltip = {
			text = rol_mars_quest_on_cooldown_tt
			NOT = { has_character_modifier = rol_mars_quest_cooldown_modifier }
		}
	}

	is_valid = {
		scope:host = {
			is_playable_character = yes
			is_alive = yes
			is_imprisoned = no
			NOT = { is_incapable = yes }
			# Settled quests stay valid until their one-day phase finish.
			OR = {
				scope:activity = { var:rol_relic_quest_closed = 1 }
				AND = {
					has_character_flag = rol_heard_mars_legend
					NOT = { has_character_flag = rol_mars_blade_claimed }
					NOT = { has_character_modifier = rol_mars_quest_cooldown_modifier }
				}
			}
		}
	}

	on_start = {
		activity_setup_special_type_progression_effect = yes
		activity_special_type_progression_variable = { NUM = 30 }
		scope:host = {
			rol_relic_quest_begin_effect = yes
			# 清理上一次结算残留并设置本次上下文
			rol_relic_clear_recent_result_effect = yes
			rol_clear_rival_route_effect = yes
			rol_set_context_mars_weapon_effect = yes
			rol_expedition_growth_begin_effect = { ROUTE = mars }
			rol_expedition_companion_begin_effect = yes
			rol_apply_expedition_preparation_start_bonus_effect = yes

			if = {
				limit = { has_variable = rol_rome_stage_count }
				remove_variable = rol_rome_stage_count
			}
			remove_character_flag = rol_mars_guest_event_fired
		}

		add_activity_log_entry = {
			key = rol_relic_activity_start_log
			character = scope:host
			score = 10
			show_in_conclusion = yes
		}
	}

	on_invalidated = {
		scope:host = { rol_rome_quest_fail_now_effect = yes }
	}

	on_host_death = {
		scope:host = {
			rol_expedition_growth_abandon_effect = yes
			rol_clear_context_mars_weapon_effect = yes
			rol_rome_quest_abort_effect = yes
			rol_relic_quest_close_effect = yes
			# Preserve the original pact fields for the on_death archive snapshot.
		}
		invalidation_host_death_effect = yes
	}


	province_filter = landed_title
	ai_province_filter = landed_title
	province_filter_target = c_roma

	is_location_valid = {
		county = title:c_roma
	}

	is_single_location = yes

	wait_time_before_start = { days = 7 }
	max_guest_arrival_delay_time = { months = 6 }
	max_guests = 12
	allow_zero_guest_invites = yes
	open_invite = no

	guest_invite_rules = {
		rules = {
			2 = activity_invite_rule_lovers
			2 = activity_invite_rule_rivals_if_appropriate
			3 = activity_invite_rule_extended_family
			5 = activity_invite_rule_knights
			6 = activity_invite_mp
		}
		defaults = {
			1 = activity_invite_rule_friends
			1 = activity_invite_rule_close_family
			3 = activity_invite_rule_courtiers
			4 = activity_invite_rule_guests
			1 = activity_invite_spouses
		}
	}

	can_be_activity_guest = {
		is_adult = yes
		is_healthy = yes
		is_available = yes
		in_diplomatic_range = scope:host
		is_playable_character = no
	}

	guest_join_chance = {
		base = 10
		base_activity_modifier = yes
		activity_guest_shared_ai_accept_modifier = yes
	}

	options = {
		rol_option_expedition_preparation = {
			rol_expedition_preparation_low = {
				cost = {
					gold = {
						add = {
							value = 30
							desc = rol_expedition_preparation_low
						}
					}
				}
			}
			rol_expedition_preparation_standard = {
				default = yes
				cost = {
					gold = {
						add = {
							value = 150
							desc = rol_expedition_preparation_standard
						}
					}
				}
			}
			rol_expedition_preparation_high = {
				cost = {
					gold = {
						add = {
							value = 600
							desc = rol_expedition_preparation_high
						}
					}
				}
			}
			rol_expedition_preparation_supreme = {
				cost = {
					gold = {
						add = {
							value = 1750
							desc = rol_expedition_preparation_supreme
						}
					}
				}
			}
		}
	}

	phases = {
		rol_mars_descent_phase = {
			is_predefined = yes
			order = 1

			select_scripted_location = {
				title:c_roma.title_province = { save_scope_as = province }
			}

			location_source = scripted

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					# 给足时间，避免多阶段事件因延迟而被活动结束打断
					scope:activity = { progress_activity_phase_after = { days = 365 } }
					trigger_event = rol_quest_rome.0990
				}
			}
		}
	}

	# 仅用于防止同一天重复点击；真实冷却由角色修正控制（2年）
	cooldown = { days = 1 }

	cost = {
		gold = {
			add = {
				value = 150
				desc = rol_seek_mars_blade_cost_desc
			}
		}
	}

	ui_predicted_cost = {
		gold = {
			value = 150
		}
	}

	host_intents = {
		intents = { rol_relic_hunt_intent }
		default = rol_relic_hunt_intent
	}

	guest_intents = {
		intents = { reduce_stress_intent }
		default = reduce_stress_intent
	}

	conclusion_description = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_character_flag = rol_recent_mars_quest_success
					has_variable = rol_recent_relic_result_tier
					var:rol_recent_relic_result_tier >= 4
				}
				desc = activity_rol_seek_mars_blade_conclusion_desc_success_4
			}
			triggered_desc = {
				trigger = {
					has_character_flag = rol_recent_mars_quest_success
					has_variable = rol_recent_relic_result_tier
					var:rol_recent_relic_result_tier >= 3
				}
				desc = activity_rol_seek_mars_blade_conclusion_desc_success_3
			}
			triggered_desc = {
				trigger = {
					has_character_flag = rol_recent_mars_quest_success
					has_variable = rol_recent_relic_result_tier
					var:rol_recent_relic_result_tier >= 2
				}
				desc = activity_rol_seek_mars_blade_conclusion_desc_success_2
			}
			triggered_desc = {
				trigger = {
					has_character_flag = rol_recent_mars_quest_success
				}
				desc = activity_rol_seek_mars_blade_conclusion_desc_success_1
			}
			triggered_desc = {
				trigger = {
					has_character_flag = rol_recent_mars_quest_failure
				}
				desc = activity_rol_seek_mars_blade_conclusion_desc_failure
			}
			desc = activity_rol_seek_mars_blade_conclusion_desc_fallback
		}
	}

	on_complete = {
		scope:host = {
			rol_rome_quest_fail_now_effect = yes
			rol_expedition_companion_finish_effect = { STAGE_VARIABLE = rol_rome_stage_count }
			rol_expedition_growth_abandon_effect = yes

			rol_clear_context_mars_weapon_effect = yes
			if = {
				limit = { has_variable = rol_rome_stage_count }
				remove_variable = rol_rome_stage_count
			}
		}
	}

	header_background = "gfx/interface/illustrations/activity_header_backgrounds/activity_rol_seek_mars_blade.dds"

	background = {
		trigger = {
			rol_activity_host_at_activity_location_trigger = yes
		}
		texture = "gfx/interface/illustrations/event_scenes/rol_activity_rome_location.dds"
		environment = "environment_event_dungeon"

	}

	# 行军途中：回廊背景
	background = {
		texture = "gfx/interface/illustrations/event_scenes/rol_activity_rome_travel.dds"
		environment = "environment_event_corridor"
		ambience = "event:/SFX/Events/Backgrounds/castle_corridor_night"
	}

	# 活动窗口人物（参考原版活动，保证背景中有角色演出）
	window_characters = {
		host = {
			camera = camera_event_center

			effect = {
				if = {
					limit = { rol_activity_host_at_activity_location_trigger = yes }
					scope:host = { add_to_list = characters }
				}
			}

			scripted_animation = {
				animation = idle
			}
		}

		travel_host = {
			camera = camera_event_horse_left

			effect = {
				if = {
					limit = { rol_activity_host_traveling_trigger = yes }
					scope:host = { add_to_list = characters }
				}
			}

			scripted_animation = {
				animation = jockey_walk
			}
		}

		guest = {
			camera = camera_body_right

			effect = {
				if = {
					limit = {
						rol_activity_host_at_activity_location_trigger = yes
						scope:player ?= { this != scope:host }
					}
					scope:player ?= { add_to_list = characters }
				}
				else = {
					if = {
						limit = { rol_activity_host_at_activity_location_trigger = yes }
						every_attending_character = {
							limit = {
								location = scope:activity.activity_location
								this != scope:host
							}
							add_to_list = characters
						}
					}
				}
			}

			scripted_animation = {
				animation = idle
			}
		}
	}

	# 活动内成功条插件
	activity_window_widgets = {
		rol_mars_success_bar = "activity_plugin_widgets_summary"
	}

	ai_will_do = {
		value = 0
	}
	ai_check_interval = 0
