# Mythic Relics - 纳克索斯：爱琴海誓卫远征
activity_rol_seek_naxos_guard = {
	is_shown = {
		is_playable_character = yes
		is_ai = no
		has_character_flag = rol_heard_naxos_legend
		NOT = { has_character_flag = rol_naxos_unit_unlocked }
	}

	can_start_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
		gold >= 230
		NOT = { is_activity_type_on_cooldown = activity_rol_seek_naxos_guard }
		custom_tooltip = {
			text = rol_naxos_quest_on_cooldown_tt
			NOT = { has_character_modifier = rol_naxos_quest_cooldown_modifier }
		}
	}

	is_valid = {
		scope:host = {
			is_playable_character = yes
			is_ai = no
			is_alive = yes
			is_imprisoned = no
			NOT = { is_incapable = yes }
			has_character_flag = rol_heard_naxos_legend
			NOT = { has_character_flag = rol_naxos_unit_unlocked }
			NOT = { has_character_modifier = rol_naxos_quest_cooldown_modifier }
		}
	}

	on_start = {
		scope:host = {
			rol_relic_clear_recent_result_effect = yes
			rol_clear_rival_route_effect = yes
			rol_set_context_naxos_relic_effect = yes
			if = {
				limit = { has_variable = rol_naxos_stage_count }
				remove_variable = rol_naxos_stage_count
			}
			remove_character_flag = rol_naxos_guest_event_fired
		}

		activity_setup_special_type_progression_effect = yes
		activity_special_type_progression_variable = { NUM = 30 }

		add_activity_log_entry = {
			key = rol_relic_activity_start_log
			character = scope:host
			score = 10
			show_in_conclusion = yes
		}
	}

	province_filter = landed_title
	ai_province_filter = landed_title
	province_filter_target = b_thera

	is_location_valid = {
		OR = {
			barony = title:b_thera
			county = title:c_naxos
		}
	}

	wait_time_before_start = { days = 7 }
	max_guest_arrival_delay_time = { months = 6 }
	max_guests = 12
	allow_zero_guest_invites = yes
	open_invite = no

	guest_invite_rules = {
		rules = {
			2 = activity_invite_rule_lovers
			2 = activity_invite_rule_rivals_if_appropriate
			3 = activity_invite_rule_extended_family
			5 = activity_invite_rule_knights
			6 = activity_invite_mp
		}
		defaults = {
			1 = activity_invite_rule_friends
			1 = activity_invite_rule_close_family
			3 = activity_invite_rule_courtiers
			4 = activity_invite_rule_guests
			1 = activity_invite_spouses
		}
	}

	can_be_activity_guest = {
		is_adult = yes
		is_healthy = yes
		is_available = yes
		in_diplomatic_range = scope:host
		is_playable_character = no
	}

	guest_join_chance = {
		base = 10
		base_activity_modifier = yes
		activity_guest_shared_ai_accept_modifier = yes
	}

	options = {
		rol_option_expedition_preparation = {
			rol_expedition_preparation_low = {
				cost = {
					gold = {
						add = {
							value = 30
							desc = rol_expedition_preparation_low
						}
					}
				}
			}
			rol_expedition_preparation_standard = {
				default = yes
				cost = {
					gold = {
						add = {
							value = 150
							desc = rol_expedition_preparation_standard
						}
					}
				}
			}
			rol_expedition_preparation_high = {
				cost = {
					gold = {
						add = {
							value = 600
							desc = rol_expedition_preparation_high
						}
					}
				}
			}
			rol_expedition_preparation_supreme = {
				cost = {
					gold = {
						add = {
							value = 1750
							desc = rol_expedition_preparation_supreme
						}
					}
				}
			}
		}
	}

	phases = {
		rol_naxos_descent_phase = {
			is_predefined = yes
			order = 1

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { days = 365 } }
					trigger_event = rol_quest_naxos.0990
				}
			}
		}
	}

	cooldown = { days = 1 }

	cost = {
		gold = {
			add = {
				value = 200
				desc = rol_seek_naxos_guard_cost_desc
			}
		}
	}

	ui_predicted_cost = {
		gold = {
			value = 200
		}
	}

	host_intents = {
		intents = { rol_relic_hunt_intent }
		default = rol_relic_hunt_intent
	}

	guest_intents = {
		intents = { reduce_stress_intent }
		default = reduce_stress_intent
	}

	conclusion_description = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_character_flag = rol_recent_naxos_quest_success
					has_variable = rol_recent_relic_result_tier
					var:rol_recent_relic_result_tier >= 4
				}
				desc = activity_rol_seek_naxos_guard_conclusion_desc_success_4
			}
			triggered_desc = {
				trigger = {
					has_character_flag = rol_recent_naxos_quest_success
					has_variable = rol_recent_relic_result_tier
					var:rol_recent_relic_result_tier >= 3
				}
				desc = activity_rol_seek_naxos_guard_conclusion_desc_success_3
			}
			triggered_desc = {
				trigger = {
					has_character_flag = rol_recent_naxos_quest_success
					has_variable = rol_recent_relic_result_tier
					var:rol_recent_relic_result_tier >= 2
				}
				desc = activity_rol_seek_naxos_guard_conclusion_desc_success_2
			}
			triggered_desc = {
				trigger = { has_character_flag = rol_recent_naxos_quest_success }
				desc = activity_rol_seek_naxos_guard_conclusion_desc_success_1
			}
			triggered_desc = {
				trigger = { has_character_flag = rol_recent_naxos_quest_failure }
				desc = activity_rol_seek_naxos_guard_conclusion_desc_failure
			}
			desc = activity_rol_seek_naxos_guard_conclusion_desc_fallback
		}
	}

	on_complete = {
		scope:host = {

			if = {
				limit = { has_variable = rol_naxos_stage_count }
				remove_variable = rol_naxos_stage_count
			}
			rol_clear_context_naxos_relic_effect = yes
		}
	}

	header_background = "gfx/interface/illustrations/activity_header_backgrounds/activity_rol_seek_naxos_guard.dds"

	background = {
		trigger = {
			rol_activity_host_at_activity_location_trigger = yes
		}
		texture = "gfx/interface/illustrations/event_scenes/rol_activity_naxos_location.dds"
		environment = "environment_event_dungeon"
		ambience = "event:/SFX/Events/Backgrounds/dungeon"
	}

	background = {
		texture = "gfx/interface/illustrations/event_scenes/rol_activity_naxos_travel.dds"
		environment = "environment_event_corridor"
		ambience = "event:/SFX/Events/Backgrounds/castle_corridor_night"
	}

	window_characters = {
		host = {
			camera = camera_event_center

			effect = {
				if = {
					limit = { rol_activity_host_at_activity_location_trigger = yes }
					scope:host = { add_to_list = characters }
				}
			}

			scripted_animation = {
				animation = idle
			}
		}

		travel_host = {
			camera = camera_event_horse_left

			effect = {
				if = {
					limit = { rol_activity_host_traveling_trigger = yes }
					scope:host = { add_to_list = characters }
				}
			}

			scripted_animation = {
				animation = jockey_walk
			}
		}

		guest = {
			camera = camera_body_right

			effect = {
				if = {
					limit = {
						rol_activity_host_at_activity_location_trigger = yes
						scope:player ?= { this != scope:host }
					}
					scope:player ?= { add_to_list = characters }
				}
				else = {
					if = {
						limit = { rol_activity_host_at_activity_location_trigger = yes }
						every_attending_character = {
							limit = {
								location = scope:activity.activity_location
								this != scope:host
							}
							add_to_list = characters
						}
					}
				}
			}

			scripted_animation = {
				animation = idle
			}
		}
	}

	activity_window_widgets = {
		rol_naxos_success_bar = "activity_plugin_widgets_summary"
	}

	ai_will_do = {
		value = 0
	}
	ai_check_interval = 0
}
