# Relics of Legends - short fragment expeditions

rol_seek_mars_fragment_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_smith.dds" }
	ai_check_interval = 0
	cooldown = { years = 2 }
	is_shown = {
		is_playable_character = yes
		is_ai = no
		has_character_flag = rol_mars_blade_claimed
		NOT = { has_character_flag = rol_mars_fragment_2 }
		any_character_artifact = { artifact_type = rol_blade_of_mars }
	}
	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
		gold >= 120
	}
	minimum_cost = { gold = 120 }
	effect = {
		add_character_flag = rol_fragment_route_mars
		trigger_event = { id = rol_relic_fragment.0001 days = 1 }
	}
	ai_potential = { always = no }
}
