# Relics of Legends - short fragment expeditions

rol_seek_mars_fragment_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_smith.dds" }
	ai_check_interval = 0
	cooldown = { years = 2 }
	is_shown = {
		is_playable_character = yes
		is_ai = no
		has_character_flag = rol_mars_blade_claimed
		any_character_artifact = {
			artifact_type = rol_blade_of_mars
			NOT = { has_artifact_modifier = rol_blade_fragment_2_modifier }
		}
	}
	is_valid_showing_failures_only = {
		is_available_adult = yes
		NOT = { rol_fragment_any_route_trigger = yes }
		custom_description = {
			text = rol_fragment_finish_resonance_first_tt
			NOT = { has_character_flag = rol_fragment_upgrade_pending }
		}
		is_at_war = no
		gold >= 120
	}
	cost = { gold = 120 }
	effect = {
		rol_relic_fragment_begin_effect = { ROUTE = mars TYPE = rol_blade_of_mars LAST_MODIFIER = rol_blade_fragment_2_modifier }
	}
	ai_potential = { always = no }
}
