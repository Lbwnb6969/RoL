# Relics of Legend - Unified expedition preparation values

rol_expedition_preparation_success_chance_modifier = {
	add = rol_expedition_journal_success_chance_modifier
	add = rol_expedition_companion_scholar_chance_modifier
	if = {
		limit = { rol_expedition_preparation_supreme_trigger = yes }
		add = 20
	}
	else_if = {
		limit = { rol_expedition_preparation_high_trigger = yes }
		add = 10
	}
	else_if = {
		limit = { rol_expedition_preparation_low_trigger = yes }
		add = -10
	}
}

# Fragment expedition dynamic challenge odds (attribute + Relic Seeker expertise)
rol_fragment_learning_success_chance_value = {
	value = 55
	if = { limit = { learning >= 15 } add = 15 }
	if = { limit = { learning >= 22 } add = 15 }
	if = {
		limit = {
			has_trait = rol_relic_seeker
			has_trait_xp = { trait = rol_relic_seeker track = rol_research value >= 60 }
		}
		add = 10
	}
	min = 20
	max = 95
}
