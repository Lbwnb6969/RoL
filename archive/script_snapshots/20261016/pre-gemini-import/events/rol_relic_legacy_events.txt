namespace = rol_relic_legacy

# First holder event: the relic asks its owner to embody its legend.
rol_relic_legacy.0001 = {
	type = character_event title = rol_relic_legacy.0001.t desc = rol_relic_legacy.0001.desc theme = stewardship
	trigger = { is_alive = yes is_ai = no rol_has_true_artifact_relic_trigger = yes NOT = { has_character_flag = rol_relic_legacy_duty_cooldown } }
	immediate = { random_character_artifact = { limit = { OR = { artifact_type = rol_blade_of_mars artifact_type = rol_nibelung_star_sphere artifact_type = rol_astrolabe_of_ocean artifact_type = rol_ash_forged_mask } } save_scope_as = rol_legacy_artifact } add_character_flag = { flag = rol_relic_legacy_duty_cooldown years = 5 } rol_legacy_recurring_begin_effect = { NAME = duty } }
	option = {
		name = rol_relic_legacy.0001.a flavor = rol_relic_legacy.0001.a.flavor
		trigger = { rol_legacy_duty_choice_trigger = yes }
		if = {
			limit = { rol_legacy_duty_choice_trigger = yes }
			add_prestige = 100 add_stress = 15
			rol_legacy_recurring_close_effect = { NAME = duty }
		}
	}
	option = {
		name = rol_relic_legacy.0001.b flavor = rol_relic_legacy.0001.b.flavor
		trigger = { rol_legacy_duty_choice_trigger = yes }
		if = {
			limit = { rol_legacy_duty_choice_trigger = yes }
			add_piety = 75 add_stress = -15
			rol_legacy_recurring_close_effect = { NAME = duty }
		}
	}
	option = {
		name = rol_legacy_unavailable_option
		flavor = rol_legacy_unavailable_flavor
		trigger = { NOT = { rol_legacy_duty_choice_trigger = yes } }
		rol_legacy_recurring_close_effect = { NAME = duty }
	}
}

rol_relic_legacy.0003 = {

  reference = rol_bg_relic_legacy_transfer
 }

  rol_legacy_selected_artifact_owned_trigger = yes
  NOT = {
   has_character_flag = rol_relic_legacy_transition_cooldown
  }
 }

  if = {
   limit = { rol_legacy_selected_artifact_owned_trigger = yes NOT = { has_character_flag = rol_relic_legacy_transition_cooldown } }

  trigger = {
   rol_legacy_transition_choice_trigger = yes
  }
  if = {
   limit = {
    rol_legacy_transition_choice_trigger = yes
   }
   rol_relic_legacy_clear_attitudes_effect = yes
   add_prestige = 75
   add_character_modifier = {
    modifier = rol_relic_legacy_public_modifier
    years = 5

    limit = {
     scope:rol_legacy_artifact.var:rol_relic_legacy_transfer_count >= 2

    }
   }

  }
 }

   rol_legacy_transition_choice_trigger = yes
   gold >= 100
