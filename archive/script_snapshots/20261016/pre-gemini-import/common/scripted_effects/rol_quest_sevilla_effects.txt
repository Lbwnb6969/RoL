# Mythic Relics - 塞维利亚大洋统御星盘任务效果
# 使用 rol_sevilla_stage_count 跟踪阶段推进次数

rol_sevilla_quest_progress_effect = {
	if = {
		limit = { rol_sevilla_quest_active_trigger = yes }
		if = {
			limit = { NOT = { has_variable = rol_sevilla_stage_count } }
			set_variable = {
				name = rol_sevilla_stage_count
				value = 0
			}
		}

		change_variable = {
			name = rol_sevilla_stage_count
			add = 1
		}

		rol_relic_log_stage_effect = yes
		rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_astrolabe_stage_progress }

		if = {
			limit = {
				NOT = { has_character_flag = rol_astrolabe_guest_event_fired }
				exists = scope:activity
				scope:activity = {
					any_attending_character = {
						this != root
					}
				}
				var:rol_sevilla_stage_count >= 1
				var:rol_sevilla_stage_count < 2
			}
			add_character_flag = rol_astrolabe_guest_event_fired
			add_character_flag = rol_rival_route_astrolabe
			rol_try_rival_interlude_effect = { FALLBACK_EVENT = rol_quest_sevilla.1050 }
		}

		if = {
			limit = { var:rol_sevilla_stage_count < 2 }
			trigger_event = { id = rol_quest_sevilla.1002 days = 14 }
		}
		else = {
			trigger_event = { id = rol_quest_sevilla.1001 days = 10 }
		}
	}
}

rol_sevilla_clamp_success_progress_effect = {
	scope:activity ?= {
		if = {
			limit = { var:activity_special_type_progression < 0 }
			set_variable = {
				name = activity_special_type_progression
				value = 0
			}
		}
		else_if = {
			limit = { var:activity_special_type_progression > 100 }
			set_variable = {
				name = activity_special_type_progression
				value = 100
			}
		}
	}
}

rol_sevilla_apply_tiered_bonus_reward_effect = {
	if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 55
		}
		add_prestige = 350
		add_gold = 150
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 45
		}
		add_prestige = 220
		add_gold = 80
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 35
		}
		add_prestige = 120
		add_gold = 40
	}
	else = {
		add_prestige = 60
	}
}

rol_sevilla_store_success_result_effect = {
	if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 55
		}
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 4
		}
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 45
		}
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 3
		}
	}
	else_if = {
		limit = {
			exists = scope:activity
			scope:activity.var:activity_special_type_progression >= 35
		}
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 2
		}
	}
	else = {
		set_variable = {
			name = rol_recent_relic_result_tier
			value = 1
		}
	}

	add_character_flag = rol_recent_sevilla_quest_success
	remove_character_flag = rol_recent_sevilla_quest_failure
}

rol_sevilla_store_failure_result_effect = {
	set_variable = {
		name = rol_recent_relic_result_tier
		value = 0
	}
	add_character_flag = rol_recent_sevilla_quest_failure
	remove_character_flag = rol_recent_sevilla_quest_success
}

rol_sevilla_quest_success_finalize_effect = {
	if = {
		limit = { rol_sevilla_quest_active_trigger = yes }
		rol_expedition_companion_finish_effect = { STAGE_VARIABLE = rol_sevilla_stage_count }
		rol_expedition_growth_finish_effect = { ROUTE = astrolabe RESULT_XP = 30 LEAVE_NOTES = no }
		rol_sevilla_clamp_success_progress_effect = yes
		rol_sevilla_store_success_result_effect = yes
		rol_sevilla_apply_tiered_bonus_reward_effect = yes
		rol_apply_expedition_preparation_success_bonus_effect = yes
		rol_store_recent_expedition_preparation_effect = yes

		add_character_flag = rol_astrolabe_of_ocean_claimed
		rol_check_legend_circle_revelation_effect = yes
		rol_check_legend_circle_fivefold_revelation_effect = yes
		remove_character_flag = rol_heard_astrolabe_legend
		remove_character_modifier = rol_astrolabe_quest_cooldown_modifier

		rol_relic_log_success_effect = yes
		send_interface_toast = {
			type = event_toast_effect_good
			title = rol_astrolabe_activity_success_toast_title
			desc = rol_astrolabe_activity_success_toast_desc
			left_icon = this
		}

		trigger_event = { id = rol_quest_sevilla.1090 days = 1 }

		rol_clear_context_astrolabe_accessory_effect = yes
		rol_sevilla_quest_abort_effect = yes
		rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_astrolabe_success_finalize }
		rol_relic_quest_close_effect = yes
	}
}

rol_sevilla_quest_partial_finalize_effect = {
	if = {
		limit = { rol_sevilla_quest_active_trigger = yes }
		rol_expedition_companion_finish_effect = { STAGE_VARIABLE = rol_sevilla_stage_count }
		rol_expedition_growth_finish_effect = { ROUTE = astrolabe RESULT_XP = 20 LEAVE_NOTES = yes }
		rol_sevilla_clamp_success_progress_effect = yes
		rol_sevilla_store_success_result_effect = yes
		rol_sevilla_apply_tiered_bonus_reward_effect = yes
		rol_store_recent_expedition_preparation_effect = yes
		rol_relic_partial_success_finalize_effect = {
			PARTIAL_MODIFIER = rol_partial_relic_evidence_modifier
			COOLDOWN_MODIFIER = rol_astrolabe_quest_cooldown_modifier
			SUMMARY_EVENT = rol_quest_sevilla.1090
			DEBUG_KEY = rol_relic_debug_astrolabe_partial_finalize
		}
		rol_clear_context_astrolabe_accessory_effect = yes
		rol_sevilla_quest_abort_effect = yes
		rol_relic_quest_close_effect = yes
	}
}

rol_apply_astrolabe_activity_cooldown_effect = {
	if = {
		limit = { rol_sevilla_quest_active_trigger = yes }
		if = {
			limit = { has_variable = rol_sevilla_stage_count var:rol_sevilla_stage_count >= 1 }
			rol_expedition_growth_finish_effect = { ROUTE = astrolabe RESULT_XP = 10 LEAVE_NOTES = no }
		}
		else = { rol_expedition_growth_abandon_effect = yes }
		rol_expedition_companion_finish_effect = { STAGE_VARIABLE = rol_sevilla_stage_count }
		rol_relic_apply_cooldown_effect = {
			COOLDOWN_MODIFIER = rol_astrolabe_quest_cooldown_modifier
			COOLDOWN_YEARS = 2
		}

		rol_sevilla_store_failure_result_effect = yes
		rol_relic_log_failure_effect = yes
		send_interface_toast = {
			type = event_toast_effect_bad
			title = rol_astrolabe_activity_failure_toast_title
			desc = rol_astrolabe_activity_failure_toast_desc
			left_icon = this
		}

		trigger_event = { id = rol_quest_sevilla.1090 days = 1 }

		rol_clear_context_astrolabe_accessory_effect = yes
		rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_astrolabe_failure_finalize }
		rol_relic_quest_close_effect = yes
	}
}

rol_sevilla_quest_fail_now_effect = {
	if = {
		limit = { rol_sevilla_quest_active_trigger = yes }
		rol_apply_astrolabe_activity_cooldown_effect = yes
		rol_sevilla_quest_abort_effect = yes
		rol_relic_debug_trace_effect = { DEBUG_KEY = rol_relic_debug_astrolabe_fail_now }
	}
}

rol_sevilla_quest_abort_effect = {
	if = {
		limit = { has_variable = rol_sevilla_stage_count }
		remove_variable = rol_sevilla_stage_count
	}
}
